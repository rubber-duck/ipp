cmake_minimum_required(VERSION 3.21)
project(ipp_cef LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(PROJECT_ARCH arm64)
set(USE_SANDBOX OFF CACHE BOOL "CEF helper sandbox")
set(CEF_API_VERSION "15400" CACHE STRING "Pinned CEF API")
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" wrapper)
add_library(ipp-cef SHARED engine.mm)
add_executable(ipp-cef-helper helper.mm)
foreach(target ipp-cef ipp-cef-helper)
  target_include_directories(${target} PRIVATE "${CEF_ROOT}")
  target_compile_definitions(${target} PRIVATE CEF_API_VERSION=${CEF_API_VERSION})
  target_compile_options(${target} PRIVATE -fobjc-arc)
  target_link_libraries(${target} PRIVATE libcef_dll_wrapper "-framework AppKit")
endforeach()
